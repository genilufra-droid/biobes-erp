window.packSourceSum = function(prefix){
  try{
    const c = packSourceContainer(prefix);
    if(!c) return;
    const rows = packSourceRows(prefix);
    const totalNeeded = ($num(prefix+'Bags',0))*($num(prefix+'BagWeight',0));
    // Sasia shkruhet GJITHMONË nga përdoruesi — nuk mbushet dhe nuk mbishkruhet automatikisht.
    let sum = 0, over = [];
    rows.forEach(r=>{
      const inp = r.querySelector('[data-pack-qty]');
      const v = parseFloat(inp?.value)||0;
      sum += v;
      // paralajmërim vizual + tekst kur sasia e shkruar kalon stokun e papaketuar të lotit
      try{
        const lot = by('lots', r.dataset.lot);
        const free = pkFree(lot, packSourceOldMap(prefix));
        const warnEl = r.querySelector('[data-pack-warn]');

        if(v > free + 0.001) over.push({lot, v, free});

    sum = Math.round(sum*1000)/1000;
    const diff = Math.round((totalNeeded - sum)*1000)/1000;
    // update empty placeholder

    if(empty) empty.style.display = rows.length ? 'none' : '';
    // update KPIs

window.packSourceSum = function(prefix){
  try{
    const c = packSourceContainer(prefix);
    if(!c) return;
    const rows = packSourceRows(prefix);
    const totalNeeded = ($num(prefix+'Bags',0))*($num(prefix+'BagWeight',0));
    // Sasia shkruhet GJITHMONË nga përdoruesi — nuk mbushet dhe nuk mbishkruhet automatikisht.
    let sum = 0, over = [];
    rows.forEach(r=>{
      const inp = r.querySelector('[data-pack-qty]');
      const v = parseFloat(inp?.value)||0;
      sum += v;
      // paralajmërim vizual + tekst kur sasia e shkruar kalon stokun e papaketuar të lotit
      try{
        const lot = by('lots', r.dataset.lot);
        const free = pkFree(lot, packSourceOldMap(prefix));
        const warnEl = r.querySelector('[data-pack-warn]');
        if(inp){ inp.style.borderColor = (v > free + 0.001) ? '#c0392b' : ''; inp.style.background = (v > free + 0.001) ? '#fdecea' : ''; }
        if(warnEl){ warnEl.textContent = (v > free + 0.001) ? ('Nuk ka sasi të mjaftueshme në magazinë: të lira '+fmt2(free)+' kg, mungojnë '+fmt2(v-free)+' kg') : ''; }
        if(v > free + 0.001) over.push({lot, v, free});
      }catch(e){}
    });
    sum = Math.round(sum*1000)/1000;
    const diff = Math.round((totalNeeded - sum)*1000)/1000;
    // update empty placeholder
    const empty = document.getElementById(prefix+'Empty');
    if(empty) empty.style.display = rows.length ? 'none' : '';
    // update KPIs
    const kTotal = document.getElementById(prefix+'KpiTotal');
    const kNeed = document.getElementById(prefix+'KpiNeed');
    const kDiff = document.getElementById(prefix+'KpiDiff');
    if(kTotal) kTotal.querySelector('b').textContent = fmt2(sum)+' kg';
    if(kNeed) kNeed.querySelector('b').textContent = fmt2(totalNeeded)+' kg';
    if(kDiff){
      kDiff.querySelector('b').textContent = fmt2(diff)+' kg';
      kDiff.className = 'pk-kpi ' + (Math.abs(diff)<=0.01 ? 'good' : (Math.abs(diff) < 1 ? 'warn' : 'bad'));

    // also update hidden sum hint
    const el = c.querySelector('[data-pack-sum="'+prefix+'"]');

    }
    // banderola e paralajmërimit për stokun
    const ov = document.getElementById(prefix+'OverStock');
    if(ov){
      if(over.length){
        ov.style.display='';
        ov.innerHTML = '<b>⚠ Nuk ka sasi të mjaftueshme në magazinë për paketim.</b> '+over.map(o=>esc(o.lot.code)+': kërkohen '+fmt2(o.v)+' kg, të lira '+fmt2(o.free)+' kg').join(' · ')+'<br><small>Ruajtja me sasi mbi stokun lejohet vetëm me fjalëkalimin e adminit dhe me arsyen se ku do të shkojë magazina (regjistrohet në ditar).</small>';

    const m = l.code.match(/(\d+)$/);

// Override savePackaging to handle multi-lot with validation and stockMove
window.savePackaging = function(){
  try{
    const order = $val('pkOrder');
    const product = $val('pkProduct');
    const internal = $val('pkInternal').trim();
    const o = by('orders', order);
    if(!order || !product || !internal) return toast('Plotësoni porosinë, produktin dhe lotin BioBes');
    const bags = $num('pkBags',0);
    const bagWeight = $num('pkBagWeight',0);
    if(!(bags>0) || !(bagWeight>0)) return toast('Thasët dhe kg/thes duhet të jenë më të mëdha se zero');
    const total = Math.round(bags*bagWeight*1000)/1000;
    const rows = packSourceRows('pk');
    if(!rows.length) return toast('Zgjidhni të paktën një lot burimor');
    // collect sources
    const seen=new Set();
    const sources=[];
    const overStock=[];
    let sum=0;
    for(let r of rows){
      const lotId = r.dataset.lot;
      const lot = by('lots', lotId);
      if(!lot || !lot.id) return toast('Loti burimor nuk u gjet');
      if(seen.has(lotId)) return toast('Loti '+lot.code+' është shtuar dy herë');
      seen.add(lotId);
      const qty = parseFloat(r.querySelector('[data-pack-qty]').value)||0;
      if(!(qty>0)) return toast('Shkruani sasinë (kg) për lotin '+lot.code);
      const cap = lotUnpacked(lot);
      if(qty > cap + 0.001) overStock.push({lot, qty, cap});
      if(lot.product !== product) return toast('Loti '+lot.code+' është i produktit tjetër — përzieni vetëm lotë të së njëjtit produkt');
      const q = Math.round(qty*1000)/1000;
      sources.push({lot: lotId, qty: q});
      sum = Math.round((sum+q)*1000)/1000;
    }
    if(Math.abs(sum - total) > 0.01) return toast('Sasia nga lotet ('+fmt2(sum)+' kg) duhet të jetë '+fmt2(total)+' kg (diferenca '+fmt2(total-sum)+' kg)');
    if(overStock.length && !window._pkOverStockAuth){
      // Nuk ka sasi të mjaftueshme në magazinë → kërkohet fjalëkalimi i adminit + arsyeja
      return packOverStockAuthorize(overStock, ()=> savePackaging());
    }
    const overAuth = window._pkOverStockAuth || null; window._pkOverStockAuth = null;
    const data = {
      id:nextSeqId('PK',packs(),1,new Date().getFullYear()),
      order: order,
      customer: o.customer,
      product: product,
      internalLot: internal,
      customerLot: $val('pkCustomer'),
      template: $val('pkTemplate'),
      bags: bags,
      bagWeight: bagWeight,
      cropYear: $num('pkCrop', new Date().getFullYear()),
      bestBefore: $num('pkBest', new Date().getFullYear()+3),
      cert: $val('pkCert'),
      organic: $val('pkOrganic'),
      originType: $val('pkOriginType'),
      part: $val('pkPart'),
      origin: $val('pkOrigin'),
      date: new Date().toISOString().slice(0,10),
      sources: sources
    };
    if(overAuth){
      data.overStock = { authorizedBy: overAuth.admin, reason: overAuth.reason, at: new Date().toISOString(), by: (activeUser()&&activeUser().username)||'',
        lots: overStock.map(o=>({lot:o.lot.id, code:o.lot.code, requested:Math.round(o.qty*1000)/1000, free:Math.round(o.cap*1000)/1000, shortage:Math.round((o.qty-o.cap)*1000)/1000})) };
    }
    const r = atomicTransaction('Paketim', ()=> corePackage(data));
    if(!r || !r.ok) return toast((r&&r.error)||'Paketimi dështoi');
    if(overAuth){
      try{
        const txt = 'Paketim '+data.id+' MBI STOKUN e magazinës — autorizoi admini '+overAuth.admin+' · arsyeja: '+overAuth.reason+' · '+data.overStock.lots.map(l=>l.code+': kërkuar '+fmt2(l.requested)+' kg, të lira '+fmt2(l.free)+' kg (mungesë '+fmt2(l.shortage)+' kg)').join('; ');
        (state.events=state.events||[]).push({date:new Date().toLocaleString('sq-AL'),type:'Paketim mbi stokun',ref:data.id,text:txt});
        securityAudit('PACK_OVER_STOCK', txt);
      }catch(e){}
    }
    save();
    closeModal();
    render();
    setTimeout(()=> packagingCard(r.record.id), 80);
    toast(overAuth ? 'Paketimi u krijua me autorizim admini (mbi stokun)' : 'Paketimi u krijua');
  }catch(e){ console.error(e); toast('Gabim gjatë ruajtjes: '+e.message); }
};

  if(!lots.length) return '—';

  });
  // spec example: "3 lot (001+003)" — show count and codes joined with +

    const rows = packSourceRows('pk');
    if(!rows.length) return toast('Zgjidhni të paktën një lot burimor');

    const seen=new Set();
    const sources=[];
    const overStock=[];

      const lotId = r.dataset.lot;
      const lot = by('lots', lotId);

// Helper for Burime column: "3 lot (001+003+005)" — show last 3 chars of code
function burimeLabel(x){
  const lots = packSourceLots(x);
  if(!lots.length) return '—';
  const codes = lots.map(l=>{
    const c = (l.code||'').replace(/^B\d*/,'');
    // take last 3 or full code for readability
    return esc(l.code);
  });
  // spec example: "3 lot (001+003)" — show count and codes joined with +
  // we show: '3 lot (B1... + B1...)'
  const short = lots.map(l=>{
    const m = l.code.match(/(\d+)$/);
    return m ? m[1].slice(-3) : l.code;
  }).join('+');
  return `${lots.length} lot (${short})`;
}

window.savePackaging = function(){
  try{
    const order = $val('pkOrder');
    const product = $val('pkProduct');
    const internal = $val('pkInternal').trim();
    const o = by('orders', order);
    if(!order || !product || !internal) return toast('Plotësoni porosinë, produktin dhe lotin BioBes');
    const bags = $num('pkBags',0);
    const bagWeight = $num('pkBagWeight',0);
    if(!(bags>0) || !(bagWeight>0)) return toast('Thasët dhe kg/thes duhet të jenë më të mëdha se zero');
    const total = Math.round(bags*bagWeight*1000)/1000;
    const rows = packSourceRows('pk');
    if(!rows.length) return toast('Zgjidhni të paktën një lot burimor');
    // collect sources
    const seen=new Set();
    const sources=[];
    const overStock=[];
    let sum=0;
    for(let r of rows){
      const lotId = r.dataset.lot;
      const lot = by('lots', lotId);
      if(!lot || !lot.id) return toast('Loti burimor nuk u gjet');
      if(seen.has(lotId)) return toast('Loti '+lot.code+' është shtuar dy herë');
      seen.add(lotId);
      const qty = parseFloat(r.querySelector('[data-pack-qty]').value)||0;
      if(!(qty>0)) return toast('Shkruani sasinë (kg) për lotin '+lot.code);
      const cap = lotUnpacked(lot);
      if(qty > cap + 0.001) overStock.push({lot, qty, cap});
      if(lot.product !== product) return toast('Loti '+lot.code+' është i produktit tjetër — përzieni vetëm lotë të së njëjtit produkt');
      const q = Math.round(qty*1000)/1000;
      sources.push({lot: lotId, qty: q});
      sum = Math.round((sum+q)*1000)/1000;
    }
    if(Math.abs(sum - total) > 0.01) return toast('Sasia nga lotet ('+fmt2(sum)+' kg) duhet të jetë '+fmt2(total)+' kg (diferenca '+fmt2(total-sum)+' kg)');
    if(overStock.length && !window._pkOverStockAuth){
      // Nuk ka sasi të mjaftueshme në magazinë → kërkohet fjalëkalimi i adminit + arsyeja
      return packOverStockAuthorize(overStock, ()=> savePackaging());
    }
    const overAuth = window._pkOverStockAuth || null; window._pkOverStockAuth = null;
    const data = {
      id:nextSeqId('PK',packs(),1,new Date().getFullYear()),
      order: order,
      customer: o.customer,
      product: product,
      internalLot: internal,
      customerLot: $val('pkCustomer'),
      template: $val('pkTemplate'),
      bags: bags,
      bagWeight: bagWeight,
      cropYear: $num('pkCrop', new Date().getFullYear()),
      bestBefore: $num('pkBest', new Date().getFullYear()+3),
      cert: $val('pkCert'),
      organic: $val('pkOrganic'),
      originType: $val('pkOriginType'),
      part: $val('pkPart'),
      origin: $val('pkOrigin'),
      date: new Date().toISOString().slice(0,10),
      sources: sources
    };
    if(overAuth){
      data.overStock = { authorizedBy: overAuth.admin, reason: overAuth.reason, at: new Date().toISOString(), by: (activeUser()&&activeUser().username)||'',
        lots: overStock.map(o=>({lot:o.lot.id, code:o.lot.code, requested:Math.round(o.qty*1000)/1000, free:Math.round(o.cap*1000)/1000, shortage:Math.round((o.qty-o.cap)*1000)/1000})) };
    }
    const r = atomicTransaction('Paketim', ()=> corePackage(data));
    if(!r || !r.ok) return toast((r&&r.error)||'Paketimi dështoi');
    if(overAuth){
      try{
        const txt = 'Paketim '+data.id+' MBI STOKUN e magazinës — autorizoi admini '+overAuth.admin+' · arsyeja: '+overAuth.reason+' · '+data.overStock.lots.map(l=>l.code+': kërkuar '+fmt2(l.requested)+' kg, të lira '+fmt2(l.free)+' kg (mungesë '+fmt2(l.shortage)+' kg)').join('; ');
        (state.events=state.events||[]).push({date:new Date().toLocaleString('sq-AL'),type:'Paketim mbi stokun',ref:data.id,text:txt});
        securityAudit('PACK_OVER_STOCK', txt);
      }catch(e){}
    }
    save();
    closeModal();
    render();
    setTimeout(()=> packagingCard(r.record.id), 80);
    toast(overAuth ? 'Paketimi u krijua me autorizim admini (mbi stokun)' : 'Paketimi u krijua');
  }catch(e){ console.error(e); toast('Gabim gjatë ruajtjes: '+e.message); }
};
