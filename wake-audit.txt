/* ---------------- 1a) ID e rradhës FB (maksimumi ekzistues për vitin) ------------ */
function nextPurchaseId(){
  try{
    let y=new Date().getFullYear(),re=new RegExp('^FB-'+y+'-(\\d+)$'),max=0;
    (purchaseInvoices()||[]).forEach(f=>{let m=String(f.id||'').match(re);if(m)max=Math.max(max,+m[1])});
    return 'FB-'+y+'-'+String(max+1).padStart(4,'0');
  }catch(e){return 'FB-'+Date.now()}
}

/* ---------------- 1b) Parapagimet: alokimet dhe mbetja e përdorur --------------- */
function prepayAllocations(prepayId){
  try{let p=purchaseInvoices().find(f=>f.id===prepayId);return (p&&p.prepayAllocations)||[]}catch(e){return[]}
}
function prepayUnused(id){
  try{
    let p=purchaseInvoices().find(f=>f.id===id);
    if(!p||!p.isPrepayment)return 0;
    let total=toALL(p.total,p.currency,p.exchangeRate),used=prepayAllocations(id).reduce((a,l)=>a+(+l.amount||0),0);
    return Math.max(0,total-used);
  }catch(e){return 0}
}
function allocatePrepaymentsAgainstInvoice(invoiceId){
  try{
    let inv=purchaseInvoices().find(f=>f.id===invoiceId);
    if(!inv||inv.isPrepayment||inv.status!=='Konfirmuar')return;
    let remaining=toALL(inv.total,inv.currency,inv.exchangeRate);
    if(remaining<=.001)return;
    let preps=purchaseInvoices().filter(f=>f.supplier===inv.supplier&&f.isPrepayment&&f.status==='Konfirmuar')
      .sort((a,b)=>String(a.invoiceDate||a.date).localeCompare(String(b.invoiceDate||b.date)));
    let applied=0;
    preps.forEach(p=>{
      if(remaining<=.001)return;
      let free=prepayUnused(p.id);
      if(free<=.001)return;
      let take=Math.min(free,remaining);
      p.prepayAllocations=(p.prepayAllocations||[]).concat([{prepayId:p.id,invoiceId:inv.id,amount:Math.round(take*100)/100,date:new Date().toISOString().slice(0,10)}]);
      remaining-=take;applied+=take;
    });
    if(applied>.001){
      state.events.push({date:new Date().toLocaleString('sq-AL'),type:'Alokim parapagimi',ref:inv.id,text:inv.id+': u ul automatikisht '+fmt2(applied)+' ALL nga parapagimet e konfirmuara (FIFO)'});
      save();
    }
  }catch(e){console.error('allocatePrepaymentsAgainstInvoice',e)}
}
