// Update KPI logic
const origPackSourceSum = window.packSourceSum;
window.packSourceSum = function(prefix){
  try{
    const c = packSourceContainer(prefix);
    if(!c) return;
    const rows = packSourceRows(prefix);
    const totalNeeded = ($num(prefix+'Bags',0))*($num(prefix+'BagWeight',0));
    // if single row and needed >0, auto-fill
    if(rows.length===1 && totalNeeded>0){
      const inp = rows[0].querySelector('[data-pack-qty]');
      if(inp) inp.value = fmt2(totalNeeded);
    }
    let sum = 0;
    rows.forEach(r=>{
      const v = parseFloat(r.querySelector('[data-pack-qty]')?.value)||0;
      sum += v;
    });
    sum = Math.round(sum*1000)/1000;
    const diff = Math.round((totalNeeded - sum)*1000)/1000;
    // update empty placeholder
    const empty = document.getElementById(prefix+'Empty');
    if(empty) empty.style.display = rows.length ? 'none' : '';
    // update KPIs
    const kTotal = document.getElementById(prefix+'KpiTotal');
    const kNeed = document.getElementById(prefix+'KpiNeed');
    const kDiff = document.getElementById(prefix+'KpiDiff');
    if(kTotal) kTotal.querySelector('b').textContent = fmt2(sum)+' kg';
    if(kNeed) kNeed.querySelector('b').textContent = fmt2(totalNeeded)+' kg';
    if(kDiff){
      kDiff.querySelector('b').textContent = fmt2(diff)+' kg';
      kDiff.className = 'pk-kpi ' + (Math.abs(diff)<=0.01 ? 'good' : (Math.abs(diff) < 1 ? 'warn' : 'bad'));
    }
    // also update hidden sum hint
    const el = c.querySelector('[data-pack-sum="'+prefix+'"]');
    if(el){
      el.textContent = `Totali ${fmt2(sum)} / Duhet ${fmt2(totalNeeded)} / Diferenca ${fmt2(diff)}`;
    }
    // call original for compatibility but we already did
  }catch(e){ console.error(e); }
};

// Override savePackaging to handle multi-lot with validation and stockMove
window.savePackaging = function(){
  try{
    const order = $val('pkOrder');
    const product = $val('pkProduct');
    const internal = $val('pkInternal').trim();
    const o = by('orders', order);
    if(!order || !product || !internal) return toast('Plotësoni porosinë, produktin dhe lotin BioBes');
    const bags = $num('pkBags',0);
    const bagWeight = $num('pkBagWeight',0);
    if(!(bags>0) || !(bagWeight>0)) return toast('Thasët dhe kg/thes duhet të jenë më të mëdha se zero');
    const total = Math.round(bags*bagWeight*1000)/1000;
    const rows = packSourceRows('pk');
    if(!rows.length) return toast('Zgjidhni të paktën një lot burimor');
    // collect sources
    const seen=new Set();
    const sources=[];
    let sum=0;
    for(let r of rows){
      const lotId = r.dataset.lot;
      const lot = by('lots', lotId);
      if(!lot || !lot.id) return toast('Loti burimor nuk u gjet');
      if(seen.has(lotId)) return toast('Loti '+lot.code+' është shtuar dy herë');
      seen.add(lotId);
      const qty = parseFloat(r.querySelector('[data-pack-qty]').value)||0;
      if(!(qty>0)) return toast('Sasia për lotin '+lot.code+' duhet të jetë pozitive');
      const cap = lotUnpacked(lot);
      if(qty > cap + 0.001) return toast('Paketimi tejkalon stokun e papaketuar të lotit '+lot.code+' (të lira '+fmt2(cap)+' kg)');
      if(lot.product !== product) return toast('Loti '+lot.code+' është i produktit tjetër — përzieni vetëm lotë të së njëjtit produkt');
      const q = Math.round(qty*1000)/1000;
      sources.push({lot: lotId, qty: q});
      sum = Math.round((sum+q)*1000)/1000;
    }
    if(Math.abs(sum - total) > 0.01) return toast('Sasia nga lotet ('+fmt2(sum)+' kg) duhet të jetë '+fmt2(total)+' kg (diferenca '+fmt2(total-sum)+' kg)');
    const data = {
      id:nextSeqId('PK',packs(),1,new Date().getFullYear()),
      order: order,
      customer: o.customer,
      product: product,
      internalLot: internal,
      customerLot: $val('pkCustomer'),
      template: $val('pkTemplate'),
      bags: bags,
      bagWeight: bagWeight,
      cropYear: $num('pkCrop', new Date().getFullYear()),
      bestBefore: $num('pkBest', new Date().getFullYear()+3),
      cert: $val('pkCert'),
      organic: $val('pkOrganic'),
      originType: $val('pkOriginType'),
      part: $val('pkPart'),
      origin: $val('pkOrigin'),
      date: new Date().toISOString().slice(0,10),
      sources: sources
    };
    const r = atomicTransaction('Paketim', ()=> corePackage(data));
    if(!r || !r.ok) return toast((r&&r.error)||'Paketimi dështoi');
    save();
    closeModal();
    render();
    setTimeout(()=> packagingCard(r.record.id), 80);
    toast('Paketimi u krijua');
  }catch(e){ console.error(e); toast('Gabim gjatë ruajtjes: '+e.message); }
};

// Helper for Burime column: "3 lot (001+003+005)" — show last 3 chars of code
function burimeLabel(x){
  const lots = packSourceLots(x);
  if(!lots.length) return '—';
  const codes = lots.map(l=>{
    const c = (l.code||'').replace(/^B\d*/,'');
    // take last 3 or full code for readability
    return esc(l.code);
  });
  // spec example: "3 lot (001+003)" — show count and codes joined with +
  // we show: '3 lot (B1... + B1...)'
  const short = lots.map(l=>{
    const m = l.code.match(/(\d+)$/);
    return m ? m[1].slice(-3) : l.code;
  }).join('+');
  return `${lots.length} lot (${short})`;
}

// Ensure coreAdvanceShipment correctly deducts from each source lot (overwrite to be precise)
const origCoreAdvanceShipment = window.coreAdvanceShipment;
window.coreAdvanceShipment = function(id, next){
  const x=by('shipments',id),cur=normalizedShipmentStatus(x);
  const expected=shipmentStages[shipmentStages.indexOf(cur)+1];
  if(!x?.id||next!==expected)return coreResult(false,'Kalimi i statusit nuk është në rend');
  const missing=shipmentRequirements(x,next);
  if(missing.length)return coreResult(false,'Mungon: '+missing.join(', '));
  if(next==='Nisur'&&!x.stockDeducted){
    const orders=new Set(shipmentOrders(x));
    const allocations=packs().filter(p=>orders.has(p.order));
    const plan=new Map();let left=+x.net||0;
    if(!(left>0))return coreResult(false,'Pesha neto duhet pozitive');
    // Plan the entire shipment before changing stock. Partial shipments must not
    // deduct every package belonging to the order, or silently clamp a shortage.
    for(const pack of allocations){
      for(const lotId of (pack.sourceLots||[])){
        const lot=by('lots',lotId),reserved=plan.get(lotId)||0;
        if(!lot.id) return coreResult(false,'Loti burimor nuk u gjet');
        const free=Math.max(0,Math.min(+lot.packedNet||0,lotAvail(lot))-reserved);
        const take=Math.min(left,packLotQty(pack,lotId),free);
        if(take>0){plan.set(lotId,reserved+take);left-=take;}
      }
    }
    if(left>0.001)return coreResult(false,'Stok i paketuar i pamjaftueshëm');
    for(const [lotId,take] of plan){
      const lot=by('lots',lotId),before=lotAvail(lot);
      lot.packedNet=Math.max(0,(+lot.packedNet||0)-take);
      lot.availableNet=before-take;lot.net=lot.availableNet;
      coreMove({key:'SHIP:'+id+':'+lot.id,type:'OUT',source:'shipment',sourceId:id,
        lot:lot.id,product:lot.product,warehouse:lot.warehouse,qty:-take,
        unitCost:lot.unitCost,date:x.date,testRunId:x.testRunId});
    }
    x.stockDeducted=true;x.stockDeductedAt=new Date().toISOString();
  }
  x.status=next;return coreResult(true,'',x);
};
